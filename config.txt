import io.swagger.v3.oas.models.Components;
import io.swagger.v3.oas.models.OpenAPI;
import io.swagger.v3.oas.models.Operation;
import io.swagger.v3.oas.models.media.Content;
import io.swagger.v3.oas.models.media.MediaType;
import io.swagger.v3.oas.models.responses.ApiResponse;
import io.swagger.v3.oas.models.responses.ApiResponses;
import io.swagger.v3.oas.models.responses.ApiResponsesExtensions;
import io.swagger.v3.oas.models.responses.ApiResponsesExtensionsConstant;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;

import java.util.Map;

@Configuration
public class SwaggerConfig {

    @Bean
    public OpenAPI customOpenAPI() {
        return new OpenAPI()
                .components(new Components())
                .path("/your-api-path", getPathItem());
    }

    private io.swagger.v3.oas.models.PathItem getPathItem() {
        Operation operation = new Operation()
                .responses(new ApiResponses()
                        .addApiResponse("200", new ApiResponse().description("OK"))
                        .addApiResponse("400", customBadRequestResponse())
                        .addApiResponse("404", new ApiResponse().description("Not Found")));

        return new io.swagger.v3.oas.models.PathItem().get(operation);
    }

    private ApiResponse customBadRequestResponse() {
        Map<String, Object> extensions = Map.of(
                ApiResponsesExtensionsConstant.X_COMMON_EXTENSIONS,
                Map.of("code", 400,
                        "message", "Your custom bad request message"));

        return new ApiResponse()
                .description("Bad Request")
                .content(new Content().addMediaType(
                        "application/json",
                        new MediaType()))
                .extensions(extensions);
    }
}

from cryptography.hazmat.primitives.ciphers import Cipher, algorithms, modes
from cryptography.hazmat.backends import default_backend
import base64

def decrypt_data_with_decrypted_data_key(cipher_text, decrypted_data_key):
    try:
        # Decode base64 encoded ciphertext
        decode_base64 = base64.b64decode(cipher_text)
        
        # Create AES cipher instance in ECB mode with the provided key
        cipher = Cipher(
            algorithms.AES(decrypted_data_key),
            modes.ECB(),
            backend=default_backend()
        )
        
        # Create a decryptor object
        decryptor = cipher.decryptor()
        
        # Perform decryption
        decrypted_bytes = decryptor.update(decode_base64) + decryptor.finalize()
        
        # Convert decrypted bytes to string using UTF-8 encoding
        return decrypted_bytes.decode('utf-8')
    except Exception as e:
        # Log or print the exception as needed
        print(f"Exception occurred while decrypting with data key: {e}")
        return None

from Crypto.Cipher import AES
import base64

def decrypt_data_with_decrypted_data_key(cipher_text, decrypted_data_key):
    try:
        # Decode base64 encoded ciphertext
        decode_base64 = base64.b64decode(cipher_text)
        
        # Create AES cipher in ECB mode
        cipher = AES.new(decrypted_data_key, AES.MODE_ECB)
        
        # Perform decryption
        decrypted_bytes = cipher.decrypt(decode_base64)
        
        # Remove possible padding (PKCS5/7 padding)
        padding_length = decrypted_bytes[-1]
        decrypted_bytes = decrypted_bytes[:-padding_length]
        
        # Convert decrypted bytes to string using UTF-8 encoding
        return decrypted_bytes.decode('utf-8')
    except Exception as e:
        # Log or print the exception as needed
        print(f"Exception occurred while decrypting with data key: {e}")
        return None

def get_key_id(alias_name):
    # Create the KMS client
    kms_client = boto3.client('kms')

    # Initialize a variable to hold the Key ID
    key_id = None

    # List aliases (pagination handled by AWS SDK)
    paginator = kms_client.get_paginator('list_aliases')
    for page in paginator.paginate():
        # Iterate through the aliases in the current page
        for alias in page['Aliases']:
            if alias_name == alias['AliasName']:
                key_id = alias['TargetKeyId']
                return key_id  # Return the Key ID once found

    return key_id
